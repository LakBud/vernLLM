## GuardOptions

<TypeTable
  type={{
    version: {
      description: 'Names this set of checks. It is in every result and keys the memo.',
      type: 'string',
      required: true,
    },
    checks: {
      description: 'The checks, see Checks.',
      type: 'readonly Check[]',
      required: true,
    },
    memo: {
      description: 'Findings remembered per text. false turns it off.',
      type: 'false | { max?: number }',
      default: '{ max: 10_000 }',
    },
    key: {
      description: 'Keys the tokens, at least 32 bytes.',
      type: 'string | Uint8Array',
      default: 'random per process',
    },
    mask: {
      description: 'The text that replaces a masked value.',
      type: '(label: string) => string',
      default: '[EMAIL] style',
    },
    stream: {
      description: 'How a streamed answer is checked and released.',
      type: 'StreamOptions',
    },
    spotlight: {
      description: 'Marks untrusted text, see Spotlight.',
      type: "{ sources?: ('tool' | 'user' | 'assistant')[] }",
      default: 'off',
    },
    onError: {
      description: 'The default for checks that do not set their own.',
      type: "'closed' | 'open'",
      default: "'closed'",
    },
    logger: {
      description: 'Where the guard reports its own failures.',
      type: "Logger | 'silent'",
      default: 'console logger',
    },
  }}
/>

<TypeTable
  type={{

## Construction errors

<Callout type="warn">
  A bad option throws a `TypeError` that starts with `guardMiddleware:` and names the field, when
  the guard is built or when `update` is called. `update` keeps the old checks when it throws.
</Callout>
